  <script>
    // OneSignal Native Permission Trigger
    async function subscribeToNotifications() {
      const isInAppBrowser = /FBAN|FBAV|Instagram|LinkedInApp/i.test(navigator.userAgent);

      if (isInAppBrowser) {
        alert("To activate Revenue Broadcasts, please open this card in your main browser (tap the 3 dots '⋮' at the top right and select 'Open in Chrome' or 'Open in Safari').");
        return;
      }

      window.OneSignalDeferred = window.OneSignalDeferred || [];
      window.OneSignalDeferred.push(async function(OneSignal) {
        try {
          const permission = await OneSignal.Notifications.requestPermission();
          console.log("Push permission status:", permission);
          if (permission) {
            setAdminButtonState();
            window.location.href = "admin.html";
          }
        } catch (err) {
          console.log("OneSignal push error:", err);
          alert("Unable to request notification permission.");
        }
      });
    }

    document.addEventListener('DOMContentLoaded', async () => {
      // Show In-App Browser Warning if opened inside FB/Instagram
      const isInAppBrowser = /FBAN|FBAV|Instagram|LinkedInApp/i.test(navigator.userAgent);
      if (isInAppBrowser) {
        const fbNotice = document.getElementById('fb-browser-notice');
        if (fbNotice) fbNotice.classList.remove('hidden');
      }

      // Audio Unmute Handler
      const video = document.getElementById('intro-video');
      const audioBtn = document.getElementById('audio-toggle-btn');
      const audioText = document.getElementById('audio-text');
      const audioIcon = document.getElementById('audio-icon');

      if (video && audioBtn) {
        audioBtn.addEventListener('click', (e) => {
          e.preventDefault();
          if (video.muted) {
            video.muted = false;
            video.play();
            audioText.innerText = 'MUTE AUDIO';
            audioIcon.innerText = '🔊';
            audioBtn.classList.add('border-[#D4AF37]', 'text-[#D4AF37]');
          } else {
            video.muted = true;
            audioText.innerText = 'TAP FOR AUDIO';
            audioIcon.innerText = '🔈';
            audioBtn.classList.remove('border-[#D4AF37]', 'text-[#D4AF37]');
          }
        });
      }
    });
  </script>
